def LatinQuarter.IsDefining : {n : ℕ} → Finset (LatinQuarter.Cell n) → LatinQuarter.Square n → Prop :=

  LatinQuarter.IsLatin L ∧ ∀ (M : LatinQuarter.Square n), LatinQuarter.IsLatin M → LatinQuarter.Agrees P L M → M = L
